CREATE TABLE peer_percentiles (
    company_id TEXT NOT NULL,
    peer_group_name TEXT NOT NULL,
    metric TEXT NOT NULL,
    value REAL,
    percentile_rank REAL,
    year INTEGER,
    PRIMARY KEY (company_id, peer_group_name, metric, year)
);


-- Compare two peer groups side by side
-- Show top 5 companies in IT peer group ranked by ROE
SELECT company_id, value, percentile_rank
FROM peer_percentiles
WHERE peer_group_name = 'IT' AND metric = 'ROE' AND year = 2026
ORDER BY percentile_rank DESC
LIMIT 5;
