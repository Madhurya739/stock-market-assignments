import yaml
import pandas as pd
import numpy as np

def load_config(config_path="screener_config.yaml"):
    """Load screener thresholds from YAML config file."""
    with open(config_path, "r") as f:
        return yaml.safe_load(f)

def apply_filters(financial_ratios: pd.DataFrame, config_path="screener_config.yaml"):
    """
    Apply screener filters to financial_ratios DataFrame.
    Supports 15 metrics with special handling for D/E and ICR.
    Returns sorted DataFrame with composite_quality_score.
    """
    config = load_config(config_path)
    df = financial_ratios.copy()

    # --- Define filters ---
    filters = {
        "roe": ("min", "ROE min"),
        "de_ratio": ("max", "D/E max"),
        "fcf": ("min", "FCF min"),
        "revenue_cagr_5yr": ("min", "Revenue CAGR 5yr min"),
        "pat_cagr_5yr": ("min", "PAT CAGR 5yr min"),
        "opm": ("min", "OPM min"),
        "pe": ("max", "P/E max"),
        "pb": ("max", "P/B max"),
        "dividend_yield": ("min", "Dividend Yield min"),
        "icr": ("min", "ICR min"),
        "market_cap": ("min", "Market Cap min"),
        "net_profit": ("min", "Net Profit min"),
        "eps_cagr": ("min", "EPS CAGR min"),
        "asset_turnover": ("min", "Asset Turnover min"),
        "sales": ("min", "Sales min"),
    }

    mask = pd.Series(True, index=df.index)

    # --- Apply filters ---
    for col, (mode, key) in filters.items():
        if key not in config:
            continue
        threshold = config[key]

        if col == "de_ratio":
            # Skip Financials sector for D/E filter
            sector_mask = df["broad_sector"] != "Financials"
            mask &= (df[col] <= threshold) | ~sector_mask

        elif col == "icr":
            # Treat Debt Free label as infinity
            icr_values = df[col].replace("Debt Free", np.inf)
            mask &= icr_values.astype(float) >= threshold

        else:
            if mode == "min":
                mask &= df[col] >= threshold
            elif mode == "max":
                mask &= df[col] <= threshold

    df = df[mask].copy()

    # --- Composite Quality Score ---
    score_components = []
    for col, (mode, _) in filters.items():
        if col in df.columns and df[col].dtype != object:
            if mode == "min":
                norm = (df[col] - df[col].min()) / (df[col].max() - df[col].min() + 1e-9)
            else:  # max filter → inverse normalization
                norm = (df[col].max() - df[col]) / (df[col].max() - df[col].min() + 1e-9)
            score_components.append(norm)

    if score_components:
        df["composite_quality_score"] = pd.concat(score_components, axis=1).mean(axis=1)
    else:
        df["composite_quality_score"] = 0

    # --- Sort by composite score ---
    df = df.sort_values(by="composite_quality_score", ascending=False)

    return df

import pandas as pd
import numpy as np

def apply_preset(financial_ratios: pd.DataFrame, preset: str):
    """
    Apply one of the 6 preset screeners to the financial_ratios DataFrame.
    Returns filtered DataFrame with composite_quality_score.
    """
    df = financial_ratios.copy()

    # --- Define preset rules ---
    presets = {
        "Quality Compounder": {
            "roe": lambda x: x > 15,
            "de_ratio": lambda x: x < 1.0,
            "fcf": lambda x: x > 0,
            "revenue_cagr_5yr": lambda x: x > 10,
        },
        "Value Pick": {
            "pe": lambda x: x < 20,
            "pb": lambda x: x < 3.0,
            "de_ratio": lambda x: x < 2.0,
            "dividend_yield": lambda x: x > 1,
        },
        "Growth Accelerator": {
            "pat_cagr_5yr": lambda x: x > 20,
            "revenue_cagr_5yr": lambda x: x > 15,
            "de_ratio": lambda x: x < 2.0,
        },
        "Dividend Champion": {
            "dividend_yield": lambda x: x > 2,
            "dividend_payout": lambda x: x < 80,
            "fcf": lambda x: x > 0,
        },
        "Debt-Free Blue Chip": {
            "de_ratio": lambda x: x == 0,
            "roe": lambda x: x > 12,
            "sales": lambda x: x > 5000,  # Crore assumed
        },
        "Turnaround Watch": {
            "revenue_cagr_3yr": lambda x: x > 10,
            "fcf_latest": lambda x: x > 0,
            "de_ratio_trend": lambda x: x.diff().iloc[-1] < 0,  # declining YoY
        },
    }

    if preset not in presets:
        raise ValueError(f"Unknown preset: {preset}")

    rules = presets[preset]

    mask = pd.Series(True, index=df.index)
    for col, condition in rules.items():
        if col in df.columns:
            mask &= condition(df[col])

    df = df[mask].copy()

    # --- Composite Quality Score (simple average of normalized metrics) ---
    score_components = []
    for col in rules.keys():
        if col in df.columns and df[col].dtype != object:
            norm = (df[col] - df[col].min()) / (df[col].max() - df[col].min() + 1e-9)
            score_components.append(norm)

    if score_components:
        df["composite_quality_score"] = pd.concat(score_components, axis=1).mean(axis=1)
    else:
        df["composite_quality_score"] = 0

    df = df.sort_values(by="composite_quality_score", ascending=False)
    return df

def winsorize_and_scale(series: pd.Series) -> pd.Series:
    p10, p90 = series.quantile(0.1), series.quantile(0.9)
    clipped = series.clip(lower=p10, upper=p90)
    return 100 * (clipped - clipped.min()) / (clipped.max() - clipped.min() + 1e-9)
def compute_composite_score(df: pd.DataFrame) -> pd.DataFrame:
    metrics = {
        "ROE": 0.15, "ROCE": 0.10, "NPM": 0.10,
        "FCF_CAGR": 0.15, "CFO_PAT": 0.10, "FCF_Positive": 0.05,

        "DE_Score": 0.10, "ICR_Score": 0.05,
    }

    df_scores = pd.DataFrame(index=df.index)

    for col, weight in metrics.items():
        df_scores[col] = winsorize_and_scale(df[col]) * weight

    df["composite_quality_score"] = df_scores.sum(axis=1)
    return df

def winsorize_and_scale(series: pd.Series) -> pd.Series:

    Caps values at the 10th and 90th percentile (winsorisation),

    return 100 * (clipped - clipped.min()) / (clipped.max() - clipped.min() + 1e-9)

import pandas as pd

def run_all_presets(financial_ratios: pd.DataFrame, presets: list) -> dict:
    preset_results = {}
    for p in presets:
        # Apply preset filter
        result = apply_preset(financial_ratios, p)
        if result.empty:
            continue

        # Compute composite score
        result = compute_composite_score(result)

        # Add sector-relative score (if broad_sector exists)
        if "broad_sector" in result.columns:
            result = sector_relative_normalisation(result)

        # Keep only top 20 KPI columns + scores
        kpi_cols = [
            "company_id", "company_name", "ROE", "ROCE", "NPM",
            "FCF_CAGR", "CFO_PAT", "FCF_Positive",
            "Revenue_CAGR_5yr", "PAT_CAGR_5yr",
            "DE", "ICR_Score", "PE", "PB",
            "Dividend_Yield", "Dividend_Payout",
            "Revenue", "Revenue_CAGR_3yr", "FCF_Latest", "DE_Declining_YoY",
            "composite_quality_score"
        ]
        available_cols = [c for c in kpi_cols if c in result.columns]
        result = result[available_cols].sort_values("composite_quality_score", ascending=False)

        preset_results[p] = result

    return preset_results

import pandas as pd

def export_presets_to_excel(preset_results: dict, filename="screener_output.xlsx"):
    with pd.ExcelWriter(filename, engine="xlsxwriter") as writer:
        for preset, df in preset_results.items():
            df_sorted = df.sort_values("composite_quality_score", ascending=False)
            df_sorted.to_excel(writer, sheet_name=preset, index=False)

            workbook = writer.book
            worksheet = writer.sheets[preset]

            # Define formats
            green_fmt = workbook.add_format({"bg_color": "#C6EFCE"})
            red_fmt = workbook.add_format({"bg_color": "#FFC7CE"})

            # Apply preset-specific rules
            if preset == "Quality Compounder":
                worksheet.conditional_format("C2:C100", {"type": "cell", "criteria": ">", "value": 15, "format": green_fmt})
                worksheet.conditional_format("C2:C100", {"type": "cell", "criteria": "<=", "value": 15, "format": red_fmt})

            elif preset == "Value Pick":
                worksheet.conditional_format("H2:H100", {"type": "cell", "criteria": "<", "value": 20, "format": green_fmt})
                worksheet.conditional_format("H2:H100", {"type": "cell", "criteria": ">=", "value": 20, "format": red_fmt})

            elif preset == "Growth Accelerator":
                worksheet.conditional_format("I2:I100", {"type": "cell", "criteria": ">", "value": 20, "format": green_fmt})
                worksheet.conditional_format("I2:I100", {"type": "cell", "criteria": "<=", "value": 20, "format": red_fmt})

            elif preset == "Dividend Champion":
                worksheet.conditional_format("K2:K100", {"type": "cell", "criteria": ">", "value": 2, "format": green_fmt})
                worksheet.conditional_format("K2:K100", {"type": "cell", "criteria": "<=", "value": 2, "format": red_fmt})

            elif preset == "Debt-Free Blue Chip":
                worksheet.conditional_format("L2:L100", {"type": "cell", "criteria": "=", "value": 0, "format": green_fmt})
                worksheet.conditional_format("L2:L100", {"type": "cell", "criteria": "!=", "value": 0, "format": red_fmt})

            elif preset == "Turnaround Watch":

def compare_peer_groups(db_path="peer.db", group1="IT", group2="Pharma", year=2026):
    """
    Compare two peer groups side by side across all metrics for a given year.
    Returns a Pandas DataFrame with metrics and average percentiles.
    """

    query = """
    SELECT 
        s1.metric,
        s1.avg_percentile AS group1_avg,
        s2.avg_percentile AS group2_avg
    FROM peer_group_summary s1
    JOIN peer_group_summary s2
        ON s1.metric = s2.metric AND s1.year = s2.year
    WHERE s1.peer_group_name = ? 
      AND s2.peer_group_name = ? 
      AND s1.year = ?
    ORDER BY s1.metric;
    """

    conn = sqlite3.connect(db_path)
    df = pd.read_sql_query(query, conn, params=(group1, group2, year))
    conn.close()

    return df

def compute_composite_score(df):
    # Define weights for each metric
    metrics = {
        "ROE": 0.2,
        "ROCE": 0.2,
        "Net_Profit_Margin": 0.15,
        "DE": 0.1,
        "FCF": 0.1,
        "PAT_CAGR_5yr": 0.1,
        "Revenue_CAGR_5yr": 0.15
    }

    df_scores = pd.DataFrame(index=df.index)

    for col, weight in metrics.items():
        # Normalize each metric to 0–100
        col_norm = (df[col] - df[col].min()) / (df[col].max() - df[col].min() + 1e-9) * 100
        df_scores[col] = col_norm * weight

    df["Composite_Score"] = df_scores.sum(axis=1)
    return df

def plot_company_vs_peer(df, company_id, peer_group_name):

    - Company values as filled polygon

    company_rows = df[df["company_id"] == company_id]
    if company_rows.empty:
        print(f"No data found for company_id={company_id}")
        return
    company = company_rows.iloc[0]

        print(f"No data found for peer group={peer_group_name}")
        return

    norm_df = group_df.copy()
    for m in metrics:

    peer_avg += peer_avg[:1]

    # Peer group average polygon (dashed outline)

    # Labels
    ax.set_xticks(angles[:-1])
    ax.set_xticklabels(metrics)
    ax.set_title(f"Radar Chart: {company.get('company_name', company_id)} vs {peer_group_name} Avg")

import os
import numpy as np
import matplotlib.pyplot as plt

def plot_company_vs_peer_or_nifty(df, company_id, nifty_avg_df, export_dir="reports/radar_charts"):
    """
    Export radar chart:
    - If company has peer group → company polygon (filled) vs peer group average (dashed).
    - If no peer group → company polygon (filled) vs Nifty 100 average (dashed).
    """

    metrics = ["ROE","ROCE","Net_Profit_Margin","DE",
               "FCF","PAT_CAGR_5yr","Revenue_CAGR_5yr","Composite_Score"]

    os.makedirs(export_dir, exist_ok=True)

    company_rows = df[df["company_id"] == company_id]
    if company_rows.empty:
        print(f"No data found for company_id={company_id}")
        return
    company = company_rows.iloc[0]
    peer_group_name = company.get("peer_group")

    # Normalize helper
    def normalize(values):
        values = np.array(values, dtype=float)
        return 100 * (values - values.min()) / (values.max() - values.min() + 1e-9)

    # Case 1: Company has peer group
    if pd.notna(peer_group_name) and peer_group_name != "No peer group assigned":
        group_df = df[df["peer_group"] == peer_group_name]
        if group_df.empty:
            print(f"No data found for peer group={peer_group_name}")
            return

        # Normalize group metrics
        norm_df = group_df.copy()
        for m in metrics:
            norm_df[m] = normalize(norm_df[m])

        # Company normalized values
        company_values = []
        for m in metrics:
            col = norm_df[m]
            company_values.append(100 * (company[m] - col.min()) / (col.max() - col.min() + 1e-9))

        # Peer group average
        peer_avg = norm_df[metrics].mean().tolist()

        label_ref = f"{peer_group_name} Avg"

    # Case 2: No peer group → compare against Nifty 100 averages
    else:
        company_values = [company[m] for m in metrics]
        peer_avg = nifty_avg_df.loc[metrics, "avg_value"].tolist()

        # Normalize both sets together
        combined = np.array(company_values + peer_avg, dtype=float)
        min_val, max_val = combined.min(), combined.max()
        company_values = [(val - min_val)/(max_val-min_val+1e-9)*100 for val in company_values]
        peer_avg = [(val - min_val)/(max_val-min_val+1e-9)*100 for val in peer_avg]

        label_ref = "Nifty 100 Avg"

    # Close polygons
    company_values += company_values[:1]
    peer_avg += peer_avg[:1]
    angles = np.linspace(0, 2*np.pi, len(metrics), endpoint=False).tolist()
    angles += angles[:1]

    # Plot
    fig, ax = plt.subplots(figsize=(7,7), subplot_kw=dict(polar=True))
    ax.plot(angles, company_values, linewidth=2, label=company.get("company_name", company_id))
    ax.fill(angles, company_values, alpha=0.25)
    ax.plot(angles, peer_avg, linewidth=2, linestyle="dashed", color="red", label=label_ref)

    ax.set_xticks(angles[:-1])
    ax.set_xticklabels(metrics)
    ax.set_title(f"Radar Chart: {company.get('company_name', company_id)} vs {label_ref}")
    plt.legend(loc="upper right", bbox_to_anchor=(1.1, 1.1))

    # Export PNG
    filename = os.path.join(export_dir, f"{company['company_id']}_radar.png")
    plt.savefig(filename, dpi=300, bbox_inches="tight")
    plt.close(fig)
    print(f"Chart saved: {filename}")

      - Green for >=75
      - Yellow for 25–75
      - Red for <=25
    """

        "ROE","ROCE","Net_Profit_Margin","DE","FCF","PAT_CAGR_5yr","Revenue_CAGR_5yr","Composite_Score",
